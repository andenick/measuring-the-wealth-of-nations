\section*{Marxian and IO Measures: Production and Trade}

\subsection*{A. Revenue side}

\textbf{Marxian:}
\begin{align*}
TV^* &= GO_p + GO_t = 2000 \\
&= 1000 + 1000 \\
C^* &= M_p = 400 \\
VA^* &= TV^* - C^* = 1600 \\
&= VA_p + GO_t = 600 + 1000 \\
V^* &= W_p = 200 \\
S^* &= VA^* - V^* = 1400 \\
&= P_p + GO_t \\
&= P_p + P_t + (W_t + M_t) \\
&= 400 + 400 + (400 + 200) \\
\frac{S^*}{V^*} &= \text{rate of surplus value} \\
&= \frac{1400}{200} = 700\% \\
&= \frac{[P_p + P_t + (W_t + M_t)]}{W_p}
\end{align*}

\textbf{Input-output:}
\begin{align*}
GO &= GO_p + GO_t = 2000 \\
&= 1000 + 1000 \\
M &= M_p + M_t = 400 + 200 = 600 \\
VA &= GO - M = 1400 \\
&= VA_p + VA_t = 600 + 800 \\
W &= W_p + W_t = 200 + 400 = 600 \\
P &= P_p + P_t = 400 + 400 = 800 \\
\frac{P}{W} &= \text{profit/wage ratio} \\
&= \frac{800}{600} = 133\% \\
&= \frac{(P_p + P_t)}{(W_p + W_t)}
\end{align*}

\subsection*{B. Use side}

\textbf{Marxian:}
\begin{align*}
TP^* &= GP_p + GP_t = 2000 \\
U^* &= M_p = 400 \\
FP^* &= TP^* - U^* = 1600 \\
&= M_t + CON + I \\
&= 200 + 1000 + 400 \\
NP^* &= CONW_p = 200 \\
SP^* &= FP^* - NP^* = 1400 \\
&= M_t + CON + I - CONW_p \\
&= M_t + CONW_t + CONC + I \\
&= 200 + 400 + 400 + 400 \\
\frac{SP^*}{NP^*} &= \text{rate of exploitation} \\
&= \frac{1400}{200} = 700\% \\
&= \frac{(M_t + CONW_t + CONC + I)}{CONW_p}
\end{align*}

\textbf{Input-output:}
\begin{align*}
GP &= GP_p + GP_t = 2000 \\
M &= M_p + M_t = 400 + 200 = 600 \\
FD &= GP - M = 1400 \\
&= CON + I \\
&= 1000 + 400
\end{align*}
