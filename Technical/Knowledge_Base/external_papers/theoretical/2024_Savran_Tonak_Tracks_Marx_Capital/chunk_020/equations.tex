% Equation 12: Workers' Income Stage II (With Loan)
\text{Workers' Income II} = \text{Wages} + \text{Bank Loan} = \$300 + \$100 = \$400

% Equation 16: Aggregate Profit (with POA)

% ============================================================
% EMPIRICAL ESTIMATION FORMULAS
% ============================================================

% Equation 17: Public Price per Square Metre
PP = \frac{\text{Total Revenue}}{\text{Area Sold}}

% Equation 18: Market Price Deflation
MP_t = MP_{2021} \times \frac{\text{House Price Index}_t}{\text{House Price Index}_{2021}}

% Equation 19: Potential Market Revenue
\text{Potential Market Revenue}_t = MP_t \times \text{Area Sold}_t

% Equation 20: Profit upon Alienation (Real Estate)
\text{POA}_{\text{Real Estate}} = \text{Potential Market Revenue} - \text{Total Revenue}
% Equivalently:
\text{POA}_{\text{Real Estate}} = (MP - PP) \times \text{Area Sold}

% Equation 21: Interest Income on Loans to Worker Households
\text{Interest Income}_{\text{Workers}} = \text{Interest}_{\text{Consumer Loans}} + \text{Interest}_{\text{Credit Cards}}

% Equation 22: Net Profit from Worker Lending
\text{POA}_{\text{Banking}} = \text{Interest Income}_{\text{Workers}} \times \frac{\text{Net Income Ratio}}{\text{Total Income}}

% ============================================================
% AGGREGATE POA ESTIMATION
% ============================================================

% Equation 23: Total POA
\text{Total POA} = \text{POA}_{\text{Real Estate}} + \text{POA}_{\text{Banking}}

% Equation 24: POA as Percentage of Corporate Profit
\text{POA Ratio} = \frac{\text{Total POA}}{\text{Total Corporate Profit}} \times 100\%
